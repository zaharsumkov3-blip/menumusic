cmake_minimum_required(VERSION 3.25)
project(MenuMusic VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
if(NOT WIN32)
  message(FATAL_ERROR "This mod targets Windows x64 only. Core tests can be compiled separately.")
endif()
if(NOT DEFINED ENV{GEODE_SDK})
  message(FATAL_ERROR "Install Geode SDK v5.11.2 and set GEODE_SDK to its directory.")
endif()
add_library(${PROJECT_NAME} SHARED src/Main.cpp src/Library.cpp src/Transfer.cpp)
target_compile_definitions(${PROJECT_NAME} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(${PROJECT_NAME} PRIVATE winhttp)
add_subdirectory("$ENV{GEODE_SDK}" "${CMAKE_CURRENT_BINARY_DIR}/geode")
setup_geode_mod(${PROJECT_NAME})
if(MSVC)
  target_compile_options(${PROJECT_NAME} PRIVATE /utf-8 /W4)
endif()
